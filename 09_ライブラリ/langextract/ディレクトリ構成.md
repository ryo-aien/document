# ディレクトリ構成

```
langextract/
├── CITATION.cff                  # 引用情報
├── COMMUNITY_PROVIDERS.md        # コミュニティ提供プロバイダに関する情報
├── CONTRIBUTING.md               # コントリビューションガイドライン
├── Dockerfile                    # Docker イメージ定義
├── LICENSE                       # ライセンス情報
├── README.md                     # プロジェクト概要
├── autoformat.sh                 # コード整形用スクリプト
├── langextract.egg-info/         # パッケージメタデータ（ビルド生成物）
├── langextract_env/              # ローカル仮想環境（通常は外部作成）
├── pyproject.toml                # パッケージ設定・依存関係
├── scripts/                      # 補助スクリプト
│   ├── create_provider_plugin.py     # カスタムプロバイダ雛形生成
│   └── validate_community_providers.py # コミュニティ提供プロバイダ検証
├── tests/                        # テストコード一式
│   ├── test_live_api.py              # 外部 API 検証
│   ├── test_ollama_integration.py    # Ollama 連携検証
│   └── ...                          # スキーマ、ファクトリ、可視化など各種テスト
├── tox.ini                       # テスト環境設定
├── docs/                         # ドキュメント
│   ├── _static/                   # 画像・GIF
│   └── examples/                  # 解説付き抽出例
├── examples/                     # 実行例
│   ├── custom_provider_plugin/    # 外部プロバイダ拡張テンプレート
│   │   ├── langextract_provider_example/
│   │   │   ├── __init__.py
│   │   │   ├── provider.py
│   │   │   └── schema.py
│   │   ├── README.md
│   │   └── pyproject.toml
│   ├── notebooks/                 # ノートブック例（ロミオとジュリエットなど）
│   │   ├── romeo_juliet.jsonl
│   │   ├── romeo_juliet_extraction.ipynb
│   │   └── romeo_juliet_visualization.html
│   └── ollama/                    # ローカル LLM（Ollama）用クイックスタート
│       ├── Dockerfile
│       ├── README.md
│       ├── docker-compose.yml
│       └── quickstart.py
└── langextract/                  # ライブラリ本体
    ├── __init__.py               # 公開 API エントリポイント
    ├── py.typed                  # 型情報マーカー
    ├── _compat/                  # 既存 API 互換レイヤー
    │   ├── __init__.py
    │   ├── exceptions.py
    │   ├── inference.py
    │   ├── registry.py
    │   └── schema.py
    ├── core/                     # 中核ロジック
    │   ├── base_model.py         # 共通インターフェース
    │   ├── data.py               # データ定義
    │   ├── debug_utils.py        # デバッグ補助
    │   ├── exceptions.py         # 例外定義
    │   ├── schema.py             # スキーマ定義
    │   ├── tokenizer.py          # トークナイザ
    │   └── types.py              # 型定義
    ├── providers/                # 推論プロバイダ実装
    │   ├── builtin_registry.py   # ビルトインプロバイダ登録
    │   ├── gemini.py             # Gemini 実装
    │   ├── ollama.py             # Ollama 実装
    │   ├── openai.py             # OpenAI 実装
    │   ├── patterns.py           # プロバイダ解決パターン
    │   ├── router.py             # ルーティング処理
    │   └── schemas/              # プロバイダ専用スキーマ
    │       └── gemini.py
    ├── annotation.py             # アノテーション処理
    ├── chunking.py               # チャンク分割
    ├── data.py                   # データ関連処理
    ├── data_lib.py               # データ補助ライブラリ
    ├── exceptions.py             # パッケージ共通例外
    ├── extraction.py             # 抽出 API 実装
    ├── factory.py                # オブジェクト生成処理
    ├── inference.py              # 推論関連処理
    ├── io.py                     # 入出力処理
    ├── progress.py               # 進捗表示
    ├── prompting.py              # プロンプト関連処理
    ├── registry.py               # 登録処理
    ├── resolver.py               # 解決処理
    ├── tokenizer.py              # トークナイザ（上位 API）
    └── visualization.py          # 可視化処理
```
